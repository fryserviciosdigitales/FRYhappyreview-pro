<script>

async function login(){

  const email =
    document.getElementById(
      'email'
    ).value;

  const password =
    document.getElementById(
      'password'
    ).value;

  const response =

    await api(

      'login',

      {
        email,
        password
      }

    );

  if(
    response.success
  ){

    localStorage.setItem(
      'usuario',
      JSON.stringify(
        response.usuario
      )
    );

    if(
      response.usuario.rol ===
      'ADMIN'
    ){

      location.href =
        'admin.html';

    }

    else{

      location.href =
        'empresa.html';

    }

  }

  else{

    alert(
      response.error
    );

  }

}

</script>
